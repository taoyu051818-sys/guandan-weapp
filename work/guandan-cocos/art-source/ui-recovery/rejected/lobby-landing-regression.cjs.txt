class Vec3 {
  constructor (x = 0, y = 0, z = 0) { Object.assign(this, { x, y, z }) }
  clone () { return new Vec3(this.x, this.y, this.z) }
  static ONE = new Vec3(1, 1, 1)
}
class Node {
  static EventType = { TOUCH_START: 'down', TOUCH_END: 'up', TOUCH_CANCEL: 'cancel' }
  constructor (name) { this.name = name; this.position = new Vec3(); this.events = {}; nodes.push(this) }
  addComponent () { return { setContentSize: (width, height) => { this.size = { width, height } } } }
  setPosition (x, y, z) { this.position = typeof x === 'object' ? x.clone() : new Vec3(x, y, z) }
  setScale (x, y, z) { this.scale = typeof x === 'object' ? x.clone() : new Vec3(x, y, z) }
  on (event, callback) { this.events[event] = callback }
}
class Ui {
  constructor (parent) { this.parent = parent }
  panel (name, x, y, w, h, _style, parent = this.parent) {
    const node = new Node(name); node.parent = parent; node.setPosition(x, y, 0); node.size = { width: w, height: h }; return node
  }
  image (...args) { return this.panel(...args) }
}
